# Direct dependencies of ssw-design-analysis. Pinned versions live in
# requirements.lock.txt, which is what a reproduction should install.
#
# Recreate with:
#   python3 -m venv .venv && .venv/bin/pip install -r ssw-design-analysis/environment/requirements.lock.txt
#
# Verify with:
#   .venv/bin/python ssw-design-analysis/environment/check_environment.py

numpy
pandas
scipy
xarray
netCDF4          # NetCDF4/HDF5 backend -- without it xarray cannot open any .nc in this repo
h5netcdf         # second NetCDF backend, used as a fallback
h5py             # REQUIRED BY h5netcdf. netCDF4 cannot read a file-like object,
                 # so without h5py every streamed SNAPSI download fails to reduce
                 # while an on-disk open still succeeds -- see FAILURES.md 2026-09-17
lxml             # pandas.read_html, parsing the frozen NOAA SSW compendium
cftime           # non-standard model calendars (CMIP6, SNAPSI)
pyarrow          # parquet
scikit-learn     # ridge / gradient boosting in the predictability arm
statsmodels
matplotlib
requests         # CEDA, THREDDS, CPC downloads
zarr             # ERA5 via WeatherBench2 public GCS
gcsfs            # anonymous GCS access for the above
dask             # chunked reads of the CMIP6 / SNAPSI ensembles
fsspec           # HTTP byte-range reads of SNAPSI zg for centres OPeNDAP refuses (Meteo-France)
aiohttp          # fsspec's HTTP backend
pydap            # CEDA OPeNDAP. The ONLY way to get SNAPSI 3-D fields: full zg
                 # is ~5 TB, a 100 hPa polar-cap subset is ~33 GB (152x smaller)
pypdf            # reading open-access papers (PDF text) for source verification; pure Python
cdsapi>=0.7.7    # ECMWF Data Store (ECDS): S2S reforecasts for the forecast-reliability test
